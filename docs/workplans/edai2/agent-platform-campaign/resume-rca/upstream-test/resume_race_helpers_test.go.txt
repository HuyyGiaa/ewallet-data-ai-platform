package substrate

// Reproduction of the 2026-11-15 GKE resume failure against the real EnsureSessionActor:
// a follow-up message on the same session that arrives while the previous turn's asynchronous
// SuspendSessionActor (substrate_sandbox_transport.go scheduleSuspendSession) is in flight.

import (
	"context"
	"net/http"
	"net/http/httptest"
	"sync"
	"sync/atomic"
	"testing"
	"time"

	atev1alpha1 "github.com/agent-substrate/substrate/pkg/api/v1alpha1"
	"github.com/agent-substrate/substrate/pkg/proto/ateapipb"
	"github.com/kagent-dev/kagent/go/api/v1alpha2"
	"google.golang.org/grpc"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/runtime"
	utilruntime "k8s.io/apimachinery/pkg/util/runtime"
	"sigs.k8s.io/controller-runtime/pkg/client/fake"
)

// scriptedControl models ate-api actor states: SuspendActor moves RUNNING -> SUSPENDING -> SUSPENDED
// over suspendFor; ResumeActor moves SUSPENDED -> RESUMING -> RUNNING over resumeFor.
type scriptedControl struct {
	ateapipb.ControlClient
	mu         sync.Mutex
	status     ateapipb.Actor_Status
	resumes    atomic.Int32
	suspendFor time.Duration
	resumeFor  time.Duration
}

func (s *scriptedControl) set(st ateapipb.Actor_Status) { s.mu.Lock(); s.status = st; s.mu.Unlock() }
func (s *scriptedControl) get() ateapipb.Actor_Status    { s.mu.Lock(); defer s.mu.Unlock(); return s.status }

func (s *scriptedControl) GetActor(_ context.Context, in *ateapipb.GetActorRequest, _ ...grpc.CallOption) (*ateapipb.Actor, error) {
	return &ateapipb.Actor{Metadata: &ateapipb.ResourceMetadata{Name: in.GetActor().GetName()}, Status: s.get()}, nil
}

func (s *scriptedControl) SuspendActor(context.Context, *ateapipb.SuspendActorRequest, ...grpc.CallOption) (*ateapipb.SuspendActorResponse, error) {
	s.set(ateapipb.Actor_STATUS_SUSPENDING)
	time.AfterFunc(s.suspendFor, func() { s.set(ateapipb.Actor_STATUS_SUSPENDED) })
	return &ateapipb.SuspendActorResponse{}, nil
}

func (s *scriptedControl) ResumeActor(context.Context, *ateapipb.ResumeActorRequest, ...grpc.CallOption) (*ateapipb.ResumeActorResponse, error) {
	s.resumes.Add(1)
	s.set(ateapipb.Actor_STATUS_RESUMING)
	time.AfterFunc(s.resumeFor, func() { s.set(ateapipb.Actor_STATUS_RUNNING) })
	return &ateapipb.ResumeActorResponse{Actor: &ateapipb.Actor{Status: ateapipb.Actor_STATUS_RESUMING}}, nil
}

func reproBackend(t *testing.T, control *scriptedControl) (*SandboxAgentActorBackend, *v1alpha2.SandboxAgent) {
	t.Helper()
	scheme := runtime.NewScheme()
	utilruntime.Must(atev1alpha1.AddToScheme(scheme))
	tmpl := &atev1alpha1.ActorTemplate{ObjectMeta: metav1.ObjectMeta{
		Name: "drift-specialist-401c441e274e8282", Namespace: "ewallet-sandbox",
		Labels: map[string]string{SandboxAgentLabelKey: "drift-specialist"},
	}, Status: atev1alpha1.ActorTemplateStatus{Phase: atev1alpha1.PhaseReady}}
	kube := fake.NewClientBuilder().WithScheme(scheme).WithObjects(tmpl).Build()
	router := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) { w.WriteHeader(http.StatusOK) }))
	t.Cleanup(router.Close)
	sa := &v1alpha2.SandboxAgent{ObjectMeta: metav1.ObjectMeta{Name: "drift-specialist", Namespace: "ewallet-sandbox"}}
	return NewSandboxAgentActorBackend(&Client{ControlClient: control}, kube, router.URL), sa
}
